+  // An EMA of N bars has no value for its first N bars. With 252 daily bars, EMA 200 drew only
+  // the last 53. We fetch a longer range from the same feed, run the EMAs over it, and line the

+  var WARM_RANGE = { '5m': '1mo', '15m': '1mo', '1h': '6mo', '1d': '2y', '1wk': '10y' };
+  var warm = { key: '', ts: [], closes: [], state: 'off', extra: 0 };

+  window.tgWarmFetch = function (symbol, interval, candles) {
+    var key = symbol + '|' + interval;
+    if (warm.key === key && warm.state === 'ready') return;
+    warm = { key: key, ts: [], closes: [], state: 'loading', extra: 0 };
+    renderMaNote();
+    var range = WARM_RANGE[interval];
+    if (!range || typeof fetchYahooData !== 'function') { warm.state = 'off'; renderMaNote(); return; }
+    fetchYahooData(symbol, interval, range).then(function (longer) {
+      if (warm.key !== key) return;
+      if (!longer || !longer.length || !candles || !candles.length || longer[0].t >= candles[0].t) {
+        warm.state = 'off'; renderMaNote(); return;
+      }
+      warm.ts = longer.map(function (c) { return c.t; });
+      warm.closes = longer.map(function (c) { return c.c; });
+      warm.extra = longer.filter(function (c) { return c.t < candles[0].t; }).length;
+      warm.state = 'ready';
+      if (typeof lastSymbol !== 'undefined' && lastSymbol === symbol && currentInterval === interval &&
+          typeof lastCandles !== 'undefined' && lastCandles && typeof lastAnalysis !== 'undefined' && lastAnalysis) {
+        renderChart(lastCandles, lastAnalysis);
+      }
+      renderMaNote();
+    }, function () {
+      if (warm.key !== key) return;
+      warm.state = 'off'; renderMaNote();
+    });
+  };

+  window.tgWarmEmas = function (candles, periods) {
+    if (warm.state !== 'ready' || !candles || !candles.length) return null;
+    var sym = (typeof lastSymbol !== 'undefined') ? lastSymbol : '';
+    if (warm.key !== sym + '|' + currentInterval) return null;
+    if (!(warm.ts[0] < candles[0].t)) return null;
+    var closes = candles.map(function (c) { return c.c; });
+    return periods.map(function (p) {
+      var long = ema(warm.closes, p);
+      var byT = {};
+      for (var i = 0; i < warm.ts.length; i++) if (long[i] != null) byT[warm.ts[i]] = long[i];
+      var k = 2 / (p + 1);
+      var local = ema(closes, p);
+      var out = new Array(candles.length).fill(null);
+      for (var j = 0; j < candles.length; j++) {
+        var v = byT[candles[j].t];
+        if (v != null) { out[j] = v; continue; }
+        // a bar the longer fetch does not have (often the live bar): carry the EMA forward
+        if (j > 0 && out[j - 1] != null && closes[j] != null) { out[j] = closes[j] * k + out[j - 1] * (1 - k); continue; }
+        out[j] = local[j];
+      }
+      return out;
+    });
+  };
