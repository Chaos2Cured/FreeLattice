+  function renderMarks() {

+    var closes = candles.map(function (c) { return c.c; }).filter(function (c) { return c != null; });
+    var lo = Math.min.apply(null, closes), hi = Math.max.apply(null, closes), pad = (hi - lo) * 0.08;
+    var parts = [];
+    function say(name, v) {
+      if (v == null) return;

+      var where = Math.abs(pct) < 0.005 ? 'at the last close' : (Math.abs(pct).toFixed(1) + '% ' + (pct < 0 ? 'below' : 'above') + ' the last close');
+      var seen = (v < lo - pad || v > hi + pad) ? ' It is outside what this chart shows.' : '';
+      parts.push('Your ' + name + ' mark ' + v.toFixed(2) + ' is ' + where + '.' + seen);
+    }
+    say('buy', num(m.buy)); say('sell', num(m.sell));
+    if (num(m.buy) != null && last <= num(m.buy)) parts.push('The last close is at or under your buy mark.');
+    if (num(m.sell) != null && last >= num(m.sell)) parts.push('The last close is at or over your sell mark.');
+    st.textContent = parts.length ? sym + ': ' + parts.join(' ') : sym + ': no marks yet.';
