STEP 2. Golden-ratio (Fibonacci) bucket spreading (Kirk's first pick)
Add to memory-compound.js:

  // Spread keys evenly over 2^bits slots. Neighbouring keys land far apart.
  function fibBucket(key8, bits) {
    bits = bits || 8;
    var n = parseInt(String(key8).slice(0, 8), 16) >>> 0;
    var h = Math.imul(n, FIB32) >>> 0;
    return h >>> (32 - bits);
  }
  // Golden-angle position on a ring, for drawing and for replay order.
  function goldenSlot(i, size) { return Math.floor(((i * (PHI - 1)) % 1) * size); }
  api.fibBucket = fibBucket; api.goldenSlot = goldenSlot;

STEP 3. Replay at phi-spaced intervals inside Snowflake's existing rest step
Today armScheduler (snowflake-core.js l.2425) checks every HOUR and replays once a DAY. Layer, don't replace: keep DAY as the floor, and add a phi schedule per flake.
Add to memory-compound.js:

  // Next replay gap for a flake after its k-th replay: 1 day * phi^k, capped at 89 days.
  function phiGapMs(k) { return Math.min(86400000 * Math.pow(PHI, k || 0), 89 * 86400000); }
  function dueForReplay(flake, now) {
    var r = (flake && flake.replay) || { k: 0, t: 0 };
    return !r.t || (now - Date.parse(r.t)) >= phiGapMs(r.k);
  }
  api.phiGapMs = phiGapMs; api.dueForReplay = dueForReplay;

STEP 4. One shared canonical memory key
Add to memory-compound.js:

  // Normalize: NFKC, lower case, collapse whitespace, strip zero-width, trim, clip 2000.
  function normalize(text) {
    var s = String(text == null ? '' : text);
    try { s = s.normalize('NFKC'); } catch (e) {}
    return s.replace(/[\u200B-\u200D\uFEFF]/g, '').toLowerCase().replace(/\s+/g, ' ').trim().slice(0, 2000);
  }
  // Key = first 8 hex of the phi-salted SHA-256 when phiHashData exists (app.html l.26058),
  // else Snowflake's sha8 (core l.119). Both are SHA-256 based; record which one in key.alg.
  function memoryKey(text) {
    var n = normalize(text);
    if (typeof root.phiHashData === 'function') {
      return root.phiHashData(n).then(function (h) { return { key: String(h).slice(0, 8), alg: 'phi', norm: n }; });
    }
    var c = root.FLSnowflakeCore;
    return c.sha8(n).then(function (h) { return { key: h, alg: 'sha8', norm: n }; });
  }
  api.normalize = normalize; api.memoryKey = memoryKey;

Note for GC: phiHashData is a top-level async function in app.html, so it is reachable as a global by name; check typeof first, never redefine it. If it is not reachable from the module (for example in node smoke), sha8 is the fallback and alg says so. Pick ONE alg per device and store it in LS fl_memory_key_alg_v0 the first time, so keys never mix.

STEP 5. Save only what's new: Core and the memory bridge (Kirk: "vital")
Add to memory-compound.js a small local index (LS fl_memory_index_v0, bucketed by fibBucket):

  function loadIndex() { try { return JSON.parse(root.localStorage.getItem('fl_memory_index_v0')) || { b: {} }; } catch (e) { return { b: {} }; } }
  function saveIndex(ix) { try { root.localStorage.setItem('fl_memory_index_v0', JSON.stringify(ix)); } catch (e) { console.warn(MARK, 'index save failed', e); } }
  // Exact repeat = same key. Near repeat = token Jaccard >= 0.9 against keys in the same bucket's recent texts.
  function tokens(n) { return n.split(' ').filter(Boolean); }
  function jaccard(a, b) { var A = {}, i, inter = 0, uni; a.forEach(function (t) { A[t] = 1; }); b.forEach(function (t) { if (A[t]) inter++; }); uni = a.length + b.length - inter; return uni ? inter / uni : 1; }
  function isNew(text, store) {
    return memoryKey(text).then(function (k) {
      var ix = loadIndex(), bucket = String(fibBucket(k.key, 8)), list = ix.b[bucket] || [];
      for (var i = 0; i < list.length; i++) {
        if (list[i].store !== store) continue;
        if (list[i].key === k.key) return { fresh: false, why: 'exact', key: k.key };
        if (list[i].norm && jaccard(tokens(list[i].norm), tokens(k.norm)) >= 0.9) return { fresh: false, why: 'near', key: list[i].key };
      }
      return { fresh: true, key: k.key, k: k };
    });
  }
  function remember(k, store) {
    var ix = loadIndex(), bucket = String(fibBucket(k.key, 8));
    ix.b[bucket] = (ix.b[bucket] || []).concat([{ key: k.key, store: store, norm: k.norm.slice(0, 300), t: new Date().toISOString() }]).slice(-34);
    saveIndex(ix);
  }
  api.isNew = isNew; api.remember = remember;
