  function wrapOnce(obj, name, store, textOf) {
    if (!obj || typeof obj[name] !== 'function' || obj[name].__flMem) return false;
    var orig = obj[name];
    var wrapped = function () {
      var args = arguments, self = this, text = textOf.apply(null, args);
      if (roomShut() || !text) return orig.apply(self, args);
      return isNew(text, store).then(function (r) {
        if (!r.fresh) { console.info(MARK, store, 'skipped', r.why, r.key); notify('Already in your memory (' + r.why + ' match).'); return { skipped: true, why: r.why, key: r.key }; }
        return Promise.resolve(orig.apply(self, args)).then(function (out) { remember(r.k, store); return out; });
      });
    };
    wrapped.__flMem = true; obj[name] = wrapped; return true;
  }

STEP 7. Restart check (tree-ring canary)
Add to memory-compound.js:

  // On wake: read the last Snowflake ledger row back, recompute psi, and check where-I-stopped.
  function wakeCheck() {
    var c = root.FLSnowflakeCore;
    if (roomShut() || !c || typeof c.loadLedger !== 'function') return Promise.resolve({ ok: null, why: 'no core' });
    var book = c.loadLedger(), rows = (book && book.rows) || [];
    if (!rows.length) return Promise.resolve({ ok: true, why: 'empty' });
    var last = rows[rows.length - 1];
    return c.recomputePsi(last).then(function (psi) {
      var stop = null; try { stop = JSON.parse(root.localStorage.getItem('fl_snowflake_replay_v0')); } catch (e) {}
      var ok = psi === last.psi && !!last.lambda;
      return { ok: ok, lambda: last.lambda, psi: last.psi, got: psi, stoppedAt: stop };
    }).catch(function (e) { return { ok: false, why: String(e && e.message || e) }; });
  }
  api.wakeCheck = wakeCheck;
