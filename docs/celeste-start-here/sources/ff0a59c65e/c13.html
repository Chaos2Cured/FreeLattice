#!/usr/bin/env node
// v-connect-boot-probe-v0 (Alpha side): one shared look for Connect and Find,
// LM Studio merged as the LM Studio door, fl-connect.js shared core intact.
'use strict';

const assert = require('assert');
const fs = require('fs');
const path = require('path');
const vm = require('vm');

const flc = fs.readFileSync(path.join(__dirname, 'fl-connect.js'), 'utf8');
const lmp = fs.readFileSync(path.join(__dirname, 'local-mind-probe.js'), 'utf8');

function mkStore() {
  const s = {};
  return { s, ls: { getItem: (k) => (k in s ? s[k] : null), setItem: (k, v) => { s[k] = String(v); }, removeItem: (k) => { delete s[k]; } } };
}
function jsonRes(o) { return { ok: true, status: 200, json: async () => o }; }
function ctxWith(fetchFn) {
  const st = mkStore();
  const ctx = {
    localStorage: st.ls,
    location: { hostname: 'thelatticetree.com', protocol: 'https:', hash: '', search: '' },
    document: {
      hidden: false,
      documentElement: { getAttribute: () => 'garden', classList: { add() {}, remove() {}, contains() { return false; } } },
      getElementById: () => null, querySelectorAll: () => [], querySelector: () => null,
      createElement: () => ({ style: {}, setAttribute() {}, appendChild() {}, addEventListener() {}, classList: { add() {}, remove() {}, contains() { return false; } }, dataset: {} }),
      head: { appendChild() {} }, body: { contains: () => false, classList: { add() {}, remove() {} } },
      addEventListener() {}, removeEventListener() {}
    },
    fetch: fetchFn, setTimeout, clearTimeout, AbortController,
    navigator: { userAgent: 't' }, console, URL, CustomEvent: function (t, o) { this.type = t; this.detail = o && o.detail; },
    dispatchEvent() {}, addEventListener() {}
  };
  ctx.window = ctx;
  vm.runInNewContext(flc, ctx);
  vm.runInNewContext(lmp, ctx);
  return { ctx, st };
}

(async () => {
  assert.ok(/v-connect-boot-probe-v0/.test(flc), 'marker in shared core');
  assert.ok(/v-connect-boot-probe-v0/.test(lmp), 'marker in local-mind-probe.js');

  // 1) Connect mount + Find inside the reuse window: one look, LM Studio asked once
  const calls = [];
  const { ctx } = ctxWith(async (url) => {
    calls.push(String(url));
    if (/127\.0\.0\.1:1234\/v1\/models/.test(url)) return jsonRes({ data: [{ id: 'lmstudio-community/Meta-Llama-3.1-8B' }] });
    if (/127\.0\.0\.1:11434\/api\/tags/.test(url)) return jsonRes({ models: [{ name: 'llama3.2:latest' }] });
    throw new Error('quiet');
  });
  assert.ok(ctx.FlConnect && typeof ctx.FlConnect.bootProbe === 'function', 'bootProbe exported');
  await ctx.FlConnect.bootProbe();
  const before = calls.slice();
  const rep = await ctx.LocalMindProbe.look();
  const lmCalls = calls.filter((u) => /1234\/v1\/models/.test(u)).length;
  const olCalls = calls.filter((u) => /11434\/api\/tags/.test(u)).length;
  assert.strictEqual(lmCalls, 1, 'LM Studio asked once for Connect and Find together');
  assert.strictEqual(olCalls, 1, 'direct Ollama asked once for Connect and Find together');
  assert.ok(calls.length >= before.length, 'look() may still ask the other doors');
  const names = rep.foundList.map((f) => f.name);
  assert.ok(names.indexOf('LM Studio') !== -1 && names.indexOf('Ollama') !== -1, 'Find still lists Ollama and LM Studio');

  // 2) LM Studio tap in Connect merges as the LM Studio door
  let merged = null;
  ctx.LocalMindProbe.mergeFound = function (f) { merged = f; return f; };
  ctx.FlConnect.remember({ name: 'lmstudio-community/Meta-Llama-3.1-8B', base: 'http://127.0.0.1:1234', door: 'lmstudio', models: ['lmstudio-community/Meta-Llama-3.1-8B'] });
  assert.strictEqual(merged.name, 'LM Studio');
  assert.strictEqual(merged.url, 'http://127.0.0.1:1234/v1/models', 'same url as the DOORS entry');
  delete ctx.LocalMindProbe.mergeFound;

  // 3) Without mergeFound (older probe), the fallback entry still points at /v1/models
  ctx.FlConnect.remember({ name: 'x', base: 'http://127.0.0.1:1234', door: 'lmstudio' });
  const saved = JSON.parse(ctx.localStorage.getItem('fl_alpha_local_mind'));
  assert.strictEqual(saved.url, 'http://127.0.0.1:1234/v1/models', 'fallback keeps the LM Studio door');
