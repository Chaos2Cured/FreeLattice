function between(src, a, b) {
  const i = src.indexOf(a);
  assert.ok(i >= 0, 'missing ' + a.slice(0, 60));
  const j = src.indexOf(b, i + a.length);
  assert.ok(j > i, 'missing end ' + b.slice(0, 60));
  return src.slice(i, j);
}

function doorsFetch(calls, opts) {
  opts = opts || {};
  return async (url) => {
    calls.push(String(url));
    if (/127\.0\.0\.1:11434\/api\/tags/.test(url) && opts.ollama !== false) return jsonRes({ models: [{ name: 'llama3.2:latest' }, { name: 'qwen2.5:7b' }] });
    if (/127\.0\.0\.1:1234\/v1\/models/.test(url) && opts.lm !== false) return jsonRes({ data: [{ id: 'lmstudio-community/Meta-Llama-3.1-8B' }, { id: 'qwen2.5-coder-7b-instruct' }] });
    throw new Error('quiet');
  };
}
