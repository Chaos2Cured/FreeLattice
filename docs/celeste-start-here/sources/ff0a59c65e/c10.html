+  // v-connect-boot-probe-v0 (F2): ONE look at this computer for everyone who asks
+  // at boot (Connect, the Local AI toast, the Chat suggestion, the Bridge chip,
+  // discovery). One promise in flight; a finished look is reused for BOOT_TTL_MS.
+  // fresh:true (a person tapped Look again) skips the reuse but still joins a
+  // look that is already on its way. No timers: age is read from Date.now().
+  function bootProbe(opts) {
+    opts = opts || {};
+    if (_bootInflight) return _bootInflight;
+    if (!opts.fresh && _bootReport && (Date.now() - _bootAt) < BOOT_TTL_MS) {
+      return Promise.resolve(_bootReport);
+    }
+    var p = Promise.all([
+      probe({ fullScan: !!opts.fullScan, gesture: !!opts.gesture }).catch(function () { return null; }),
+      lmStudioLook().catch(function () { return null; })
+    ]).then(function (both) {
+      var report = both[0] || { models: [] };
+      report.lmstudio = both[1] || null;
+      report.lmstudioAsked = !!(both[1] && both[1].asked);
+      report.at = Date.now();
+      report.doors = doorsFrom(report);
+      _bootReport = report;
+      _bootAt = report.at;
+      if (_bootInflight === p) _bootInflight = null;
+      return report;
+    }, function () {
+      if (_bootInflight === p) _bootInflight = null;
+      return { models: [], doors: [], at: Date.now() };
+    });
+    _bootInflight = p;
+    return p;
+  }
