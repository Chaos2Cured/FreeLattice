+  function parseReply(type, json) {
+    if (!json || typeof json !== 'object') return '';

+  function send(mind, msgs) {
+    var id = providerIdOf(mind);
+    var p = provider(id);
+    if (!p) return Promise.reject(cloudErr('This chair names a cloud provider this garden does not know. Nothing was sent.'));
+    var model = cleanModel(mind.model);
+    if (!model) return Promise.reject(cloudErr('This chair has no model name for ' + p.name + '. Use Change to pick one. Nothing was sent.'));
+    if (!getKey(id)) {
+      return Promise.reject(cloudErr('No ' + p.name + ' key is in this browser now. Add it in the Gathering (Change, then Seat a mind by API key). Nothing was sent.'));
+    }
+    var req = buildRequest(id, model, msgs);
+    var ctrl = typeof AbortController !== 'undefined' ? new AbortController() : null;
+    var timer = setTimeout(function () { if (ctrl) ctrl.abort(); }, TIMEOUT_MS);
+    var opts = { method: 'POST', mode: 'cors', cache: 'no-store', credentials: 'omit', referrerPolicy: 'no-referrer',
+      headers: req.headers, body: JSON.stringify(req.body) };
+    if (ctrl) opts.signal = ctrl.signal;
+    return fetch(req.url, opts).then(function (res) {
+      clearTimeout(timer);
+      if (!res.ok) {
+        var s = res.status;
+        if (s === 401 || s === 403) throw cloudErr(p.name + ' did not accept the key. Check it, or Forget it and paste a new one. Nothing was invented.');
+        if (s === 429) throw cloudErr(p.name + ' asked us to slow down (a rate limit or no credit left). Nothing was invented.');
+        if (s === 400 || s === 404) throw cloudErr(p.name + ' did not take the model ' + model + '. Use Change to pick another. Nothing was invented.');
+        throw cloudErr(p.name + ' answered with an error (' + s + '). Nothing was invented.');
+      }
+      return res.json().then(function (json) {
+        var text = parseReply(req.type, json);
+        if (!text) throw cloudErr(p.name + ' was quiet. Nothing was invented.');
+        return text;
+      }, function () { throw cloudErr(p.name + ' sent something we could not read. Nothing was invented.'); });
+    }, function () {
+      clearTimeout(timer);
+      throw cloudErr('The browser could not reach ' + p.name + ' from here (offline, blocked, or it took too long). Nothing was invented.');
+    });
+  }
