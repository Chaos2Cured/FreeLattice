+function splitCompound(compound) {
+  var s = String(compound || '').trim();
+  var parts = [];
+  var buf = '';
+  var i;
+  var inAttr = false;
+  for (i = 0; i < s.length; i++) {
+    var ch = s.charAt(i);
+    if (ch === '[') inAttr = true;
+    if (ch === ']') inAttr = false;
+    if (!inAttr && i > 0 && (ch === '.' || ch === '#' || ch === '[')) {
+      if (buf) parts.push(buf);
+      buf = ch;
+    } else {
+      buf += ch;
+    }
+  }
+  if (buf) parts.push(buf);
+  return parts;
+}
