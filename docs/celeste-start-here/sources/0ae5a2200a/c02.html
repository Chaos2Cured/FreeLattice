+  // v-tree-workshop-way-back-v0: the benches' Settings button leaves a way back.
+  // settings.html?from=workshop shows one calm link; workshop.html?open=benches
+  // opens the benches again, so the round trip is one tap each way.
+  function wayBackToBenches() {
+    var q = location.search || '';
+    if (/\/settings(\.html)?$/i.test(location.pathname) && /[?&]from=workshop(&|$)/.test(q)) {
+      var main = document.querySelector('main');
+      if (main && !main.querySelector('[data-way-back-workshop]')) {
+        var back = document.createElement('a');
+        back.className = 'room-way-back';
+        back.href = 'workshop.html?open=benches';
+        back.setAttribute('data-way-back-workshop', '1');
+        back.textContent = '\u2190 Back to the Workshop benches';
+        back.addEventListener('click', function (ev) {
+          ev.preventDefault();
+          go(back.getAttribute('href'));
+        });
+        main.insertBefore(back, main.firstChild);
+      }
+    }
+    if (currentGalaxy() === 'workshop' && /[?&]open=benches(&|$)/.test(q)) {
+      setTimeout(function () {
+        if (window.GardenRooms && typeof GardenRooms.openWorkshopBenches === 'function') {
+          GardenRooms.openWorkshopBenches();
+        }
+      }, 120);
+    }
+  }
