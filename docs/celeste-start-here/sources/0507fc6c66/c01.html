STEP 3. Read path that reports what it found and whether it reached the context
Add to docs/modules/memory-compound.js (from 068m):
  // Exact key hit first, then similar (Jaccard >= 0.5) from the index; report both.
  function recall(text, opts) {
    opts = opts || {};
    return memoryKey(text).then(function (k) {
      var ix = loadIndex(), all = [], exact = [], similar = [];
      Object.keys(ix.b).forEach(function (b) { all = all.concat(ix.b[b]); });
      all.forEach(function (r) {
        if (r.key === k.key) exact.push(r);
        else if (r.norm && jaccard(tokens(r.norm), tokens(k.norm)) >= 0.5) similar.push(r);
      });
      return { key: k.key, exact: exact, similar: similar.slice(0, 5), reachedContext: false };
    });
  }
  function markReached(report, injectedText) {
    report.reachedContext = !!injectedText && report.exact.concat(report.similar).some(function (r) { return injectedText.toLowerCase().indexOf(r.norm.slice(0, 40)) !== -1; });
    console.info(MARK, 'recall', report.key, 'exact', report.exact.length, 'similar', report.similar.length, 'reached', report.reachedContext);
    return report;
  }
  api.recall = recall; api.markReached = markReached;
Use it in Snowflake's plan step: before ask('plan'), call FLMemory.recall(q); put exact/similar hits into the plan prompt; after building the prompt, markReached(report, prompt). Show in the receipt: "Memory: 1 exact, 2 similar, reached the context: yes."

  function placeRoundTable(menu, d) {
    if (!menu || !d || menu.querySelector('#flRoundTableAdd')) return;
    var b = d.createElement('button');
    b.type = 'button'; b.id = 'flRoundTableAdd'; b.className = 'fl-add-item';
    b.setAttribute('data-marker', 'v-chat-roundtable-button-v0.1'); b.setAttribute('aria-label', 'Round Table');
    var copy = d.createElement('span'); copy.className = 'fl-add-copy';
    var w = d.createElement('span'); w.className = 'fl-add-word'; w.textContent = 'Round Table';
    copy.appendChild(w); b.appendChild(copy);
    var snow = menu.querySelector('#flSnowflakeAdd');
    if (snow && snow.nextSibling) menu.insertBefore(b, snow.nextSibling); else menu.appendChild(b);
    b.addEventListener('click', function () {
      if (root.FLChatBottom && typeof root.FLChatBottom.close === 'function') root.FLChatBottom.close();
      var input = d.getElementById('chatInput');
      var q = input && input.value ? input.value.trim() : '';
      if (typeof root.switchTab === 'function') root.switchTab('round-table');
      // The app's lazy loader (app.html ~l.71730) loads round-table.js on tabChanged 'round-table'.
      if (q) { try { root.sessionStorage.setItem('fl_roundtable_prefill_v0', q); } catch (e) { console.warn('roundtable prefill', e); } }
    });
  }
